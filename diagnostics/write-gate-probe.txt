// benign language-model vocabulary experiment
const vocabulary=["THE","CAT","DRAGON","MOON","EATS","SEES","BUILDS","DREAMS","A","SMALL","RED","WORLD","."];
const tokens=[];
function weight(a,b){return ((a*7+b*11+a*b*3+7)%17)/16}
function predictNext(){if(!tokens.length)return 0;let best=0,score=-1;for(let b=0;b<vocabulary.length;b++){let s=0;tokens.forEach((t,i)=>s+=weight(t,b)*(i+2));if(s>score){score=s;best=b}}return best}
function appendToken(v){tokens.push(v);if(tokens.length>9)tokens.shift()}
